COPY EVERYTHING BELOW into Firebase Console → Firestore → Rules → Publish

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isSignedIn() {
      return request.auth != null;
    }

    function isAdmin() {
      return isSignedIn()
        && exists(/databases/$(database)/documents/admins/$(request.auth.uid))
        && (
          get(/databases/$(database)/documents/admins/$(request.auth.uid)).data.admin == 'admin'
          || get(/databases/$(database)/documents/admins/$(request.auth.uid)).data.role == 'admin'
          || get(/databases/$(database)/documents/admins/$(request.auth.uid)).data.admin == true
        );
    }

    function isContentCollection(name) {
      return name in [
        'hero',
        'about',
        'contact',
        'socials',
        'settings',
        'destinations',
        'gallery',
        'weatherCards'
      ];
    }

    match /admins/{userId} {
      allow read: if isSignedIn() && request.auth.uid == userId;
      allow create: if isSignedIn() && request.auth.uid == userId;
      allow update: if isSignedIn() && request.auth.uid == userId;
      allow delete: if false;
      allow list: if false;
    }

    match /{collection}/{document} {
      allow read: if isContentCollection(collection)
        || (isSignedIn() && collection == 'admins' && request.auth.uid == document);
      allow create, update, delete: if isContentCollection(collection) && isAdmin();
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
